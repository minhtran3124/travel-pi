# AI / Agent layer dependencies (specs/travel-agent).
#
# Split out from requirements.txt because the LangChain 1.x / LangGraph 1.x
# stack pulls a large transitive tree that must resolve together. Kept separate
# so the core mock-API install stays lean and any AI-dep conflict is isolated
# here. Pinned to current majors; loosen as we validate.
#
# NOTE: must NOT force an upgrade of fastapi/starlette/pydantic beyond the caps
# in requirements.txt (fastapi<0.137, starlette<1.3) — if a transitive dep
# demands a newer one, stop and escalate rather than nudging those caps.

# Core graph + framework.
# Floors raised 2026-09-18 (specs/stack-upgrade-langchain-1-4) to the minimums
# `deepagents 0.7.15` declares, so it can be added later without a second stack
# move: langchain>=1.4.1 (itself needs langgraph>=1.2.11 and langchain-core>=1.6.0),
# langchain-core>=1.6.3, langchain-anthropic>=1.7.2, langchain-google-genai>=4.4.0.
langgraph>=1.2.11,<2.0
langchain>=1.4.1,<2.0
langchain-core>=1.6.3,<2.0

# Deep-agent harness for the trip-research graph (specs/trip-research-deep-agent):
# skills, memory, permissions, rubric-with-tools, thread resume, large-tool-result
# offload come from here rather than being hand-rolled. Pulls langsmith>=0.12.6;
# resolves against the floors above (raised in specs/stack-upgrade-langchain-1-4).
deepagents>=0.7.15,<0.8

# Transitive via langgraph (1.2.11 itself requires >=1.1.0,<1.2), but pinned
# explicitly because we import from it:
# app/agents/retry.py uses `ToolInvocationError` and reads its `filtered_errors`
# to tell "the model passed bad arguments" apart from "an injected/runtime arg is
# missing" (#113). Neither exists before 1.0.2 — verified by unpacking the wheels:
# 1.0.0 and 1.0.1 have no `ToolInvocationError` and no `filtered_errors`, 1.0.2+
# have both. Without this floor, `langgraph>=1.0` can legally resolve to a build
# where importing the app raises ImportError at startup. There is no lockfile, so
# the range is the only guard.
langgraph-prebuilt>=1.0.2,<2.0

# langchain-core's DeterministicFakeEmbedding (used by app/core/embeddings.py
# whenever no LLM provider key is set — the default in tests/CI) calls
# numpy.random at runtime but langchain-core does not declare numpy as a
# dependency. Previously pulled in transitively by langchain-postgres/pgvector
# (removed with RAG); must be explicit now, or a clean install breaks every
# test that embeds without a key.
numpy>=1.21,<3

# Provider integrations (provider-agnostic via init_chat_model()).
langchain-openai>=0.3,<1.0
langchain-anthropic>=1.7.2,<2.0
langchain-google-genai>=4.4.0,<5.0  # provider="google_genai" (Gemini / AI Studio)

# RAG vector store (specs/travel-knowledge-rag). pgvector via langchain-postgres
# on Postgres (PGVector, cosine); InMemoryVectorStore degrade on SQLite/tests so
# the suite runs with no Postgres and no network. Re-added after the
# specs/remove-rag-feature deletion, re-scoped from places to visa + local tips.
# Verified pip-check clean with the langchain 1.3.11 / langgraph 1.2.6 stack.
langchain-postgres>=0.0.17,<0.1  # 0.0.13-0.0.15 pin langchain-core<0.4
pgvector>=0.3,<1.0

# app/rag/ingest.py chunks long prose with RecursiveCharacterTextSplitter.
# langchain 1.x dropped langchain-text-splitters from its dependency tree, so
# it must be explicit — an existing venv may still carry it as a stale leftover
# from a pre-1.0 install while a clean CI install has no such package.
langchain-text-splitters>=0.3,<1.0

# Checkpointers — SQLite local/test, Postgres in docker/prod.
# Pin >=3.0 to match langgraph-checkpoint 4.x core serializer (2.x calls a
# removed JsonPlusSerializer.dumps and breaks at runtime).
# 3.0.x pins langgraph-checkpoint<4.0, which langgraph>=1.2.11 no longer allows.
langgraph-checkpoint-sqlite>=3.1.0,<4.0
langgraph-checkpoint-postgres>=3.1.0,<4.0

# Observability.
langfuse>=3.0,<5.0

# MCP server — expose agent tools to external clients (Phase 9 / roadmap B11).
mcp>=1.2,<2.0

# Offline eval harness (MIN-58). Eval-only tooling — not imported by the runtime
# app; used by the DeepEval scripts/CI to grade agent responses with an
# OpenAI-native LLM judge. Verified not to force the fastapi/starlette/pydantic
# caps above (pip check clean at pin time).
deepeval>=4.0,<5

# Deterministic tool-call trajectory matching for the eval pipeline
# (agent-eval-pipeline task 2.2; MIT, langchain-ai/agentevals). Eval-only —
# not imported by the runtime app. Pinned to the exact version its API was
# verified against (`agentevals.trajectory.match.create_trajectory_match_evaluator`,
# modes strict/unordered/superset — see evalkit/metrics/trajectory.py for the
# strict<->superset naming gotcha). Depends only on `openevals>=0.0.20`, which
# resolves against the langchain/langgraph pins above with no forced bump;
# pulls in `langchain_core` at runtime as its own dependency, not ours —
# `evalkit/` itself never imports langchain directly (test_boundary.py).
agentevals==0.0.9
