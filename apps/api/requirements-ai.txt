# AI / Agent layer dependencies (specs/travel-agent).
#
# Split out from requirements.txt because the LangChain 1.x / LangGraph 1.x
# stack pulls a large transitive tree that must resolve together. Kept separate
# so the core mock-API install stays lean and any AI-dep conflict is isolated
# here. Pinned to current majors; loosen as we validate.
#
# NOTE: must NOT force an upgrade of fastapi/starlette/pydantic beyond the caps
# in requirements.txt (fastapi<0.137, starlette<1.3) — if a transitive dep
# demands a newer one, stop and escalate rather than nudging those caps.

# Core graph + framework.
langgraph>=1.0,<2.0
langchain>=1.0,<2.0

# langchain-core's DeterministicFakeEmbedding (used by app/core/embeddings.py
# whenever no LLM provider key is set — the default in tests/CI) calls
# numpy.random at runtime but langchain-core does not declare numpy as a
# dependency. Previously pulled in transitively by langchain-postgres/pgvector
# (removed with RAG); must be explicit now, or a clean install breaks every
# test that embeds without a key.
numpy>=1.21,<3

# Provider integrations (provider-agnostic via init_chat_model()).
langchain-openai>=0.3,<1.0
langchain-anthropic>=1.0,<2.0
langchain-google-genai>=4.0,<5.0  # provider="google_genai" (Gemini / AI Studio)

# RAG vector store (specs/travel-knowledge-rag). pgvector via langchain-postgres
# on Postgres (PGVector, cosine); InMemoryVectorStore degrade on SQLite/tests so
# the suite runs with no Postgres and no network. Re-added after the
# specs/remove-rag-feature deletion, re-scoped from places to visa + local tips.
# Verified pip-check clean with the langchain 1.3.11 / langgraph 1.2.6 stack.
langchain-postgres>=0.0.13,<0.1
pgvector>=0.3,<1.0

# Checkpointers — SQLite local/test, Postgres in docker/prod.
# Pin >=3.0 to match langgraph-checkpoint 4.x core serializer (2.x calls a
# removed JsonPlusSerializer.dumps and breaks at runtime).
langgraph-checkpoint-sqlite>=3.0,<4.0
langgraph-checkpoint-postgres>=3.0,<4.0

# Observability.
langfuse>=3.0,<5.0

# MCP server — expose agent tools to external clients (Phase 9 / roadmap B11).
mcp>=1.2,<2.0

# Offline eval harness (MIN-58). Eval-only tooling — not imported by the runtime
# app; used by the DeepEval scripts/CI to grade agent responses with an
# OpenAI-native LLM judge. Verified not to force the fastapi/starlette/pydantic
# caps above (pip check clean at pin time).
deepeval>=4.0,<5
